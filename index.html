Quick Commands
# Run tests
npm run test:happyq                  # runs suite + auto-generates report
npm run test:happyq:headed          # watch the browser
 
# Heal selectors after adding new pages
npm run heal:selectors -- --project happyq  # fix headings
npm run heal:actions   -- --project happyq  # fix buttons/tabs
npm run heal:dialogs   -- --project happyq  # fix dialog fields
 
# New project
npm run new:project -- --name myapp --src D:\myapp
 
# Regenerate docs
npm run generate:architecture         # rebuild this architecture doc
npm run report:happyq                 # rebuild test report + open
npm run serve:docs                    # serve this site at localhost:4000
